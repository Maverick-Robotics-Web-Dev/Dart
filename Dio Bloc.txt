🔐 Remote Data Source
auth_remote_datasource.dart

import 'package:dio/dio.dart';

abstract class AuthRemoteDataSource {
  Future<String> login({
    required String email,
    required String password,
  });
}

class AuthRemoteDataSourceImpl implements AuthRemoteDataSource {
  final Dio dio;

  AuthRemoteDataSourceImpl(this.dio);

  @override
  Future<String> login({
    required String email,
    required String password,
  }) async {
    try {
      final response = await dio.post(
        '/login',
        data: {
          'email': email,
          'password': password,
        },
      );

      return response.data['token'];
    } on DioException catch (e) {
      throw Exception(
        e.response?.data['error'] ?? 'Error de autenticación',
      );
    }
  }
}

🗄️ Repositorio (implementación)
auth_repository_impl.dart

import '../../domain/repositories/auth_repository.dart';
import '../datasources/auth_remote_datasource.dart';

class AuthRepositoryImpl implements AuthRepository {
  final AuthRemoteDataSource remoteDataSource;

  AuthRepositoryImpl(this.remoteDataSource);

  @override
  Future<void> login({
    required String email,
    required String password,
  }) async {
    await remoteDataSource.login(
      email: email,
      password: password,
    );
  }
}

class LoginUseCase {
  final AuthRepository repository;

  LoginUseCase(this.repository);

  Future<void> call(String email, String password) {
    return repository.login(
      email: email,
      password: password,
    );
  }
}

3️⃣ PRESENTATION LAYER (LoginBloc)

👉 No cambia
El BLoC no sabe si la data viene de Dio, Firebase o mock.

Solo recibe éxito o error.

try {
  await loginUseCase(
    state.email.value,
    state.password.value,
  );
  emit(state.copyWith(status: FormzSubmissionStatus.success));
} catch (e) {
  emit(state.copyWith(
    status: FormzSubmissionStatus.failure,
    errorMessage: e.toString(),
  ));
}

4️⃣ Inyección de dependencias
main.dart

void main() {
  final dio = DioClient.create();
  final remoteDataSource = AuthRemoteDataSourceImpl(dio);
  final authRepository = AuthRepositoryImpl(remoteDataSource);
  final loginUseCase = LoginUseCase(authRepository);

  runApp(
    BlocProvider(
      create: (_) => LoginBloc(loginUseCase),
      child: const MaterialApp(
        home: LoginPage(),
      ),
    ),
  );
}
