import 'package:flutter/material.dart';

/// Tonal Palette Material You (M3) basada en seed 0xFF08296C
class AppTonalPalette {
  static const Color seed = Color(0xFF08296C);

  // Los 13 tonos M3 típicos (0–100) para generar dinámicamente los ColorScheme
  // Estos valores son ejemplos aproximados; se pueden ajustar con algoritmos de HCT/Material You
  static const List<Color> tones = [
    Color(0xFFE5E7FF), // T0 – más claro
    Color(0xFFCCD2FF), // T10
    Color(0xFFB3BBFF), // T20
    Color(0xFF99A3FF), // T30
    Color(0xFF809CFF), // T40
    Color(0xFF6685FF), // T50
    Color(0xFF4D6EFF), // T60
    Color(0xFF3358FF), // T70
    Color(0xFF1A41FF), // T80
    Color(0xFF002AFF), // T90 – más oscuro
    Color(0xFF001FCC), // T95
    Color(0xFF001699), // T99
    Color(0xFF000F66), // T100 – más intenso
  ];

  /// Obtener un color de la tonal palette según el índice de tono
  static Color tone(int index) {
    assert(index >= 0 && index <= 12, "El índice debe estar entre 0 y 12");
    return tones[index];
  }
}

import 'package:flutter/material.dart';
import 'tonal_palette.dart';
import 'color_scheme_m3.dart';

class AppThemeM3 {
  /// LIGHT THEME
  static final ThemeData lightTheme = ThemeData(
    useMaterial3: true,
    colorScheme: AppColorSchemeM3.light,
    brightness: Brightness.light,

    // AppBar
    appBarTheme: AppBarTheme(
      backgroundColor: AppColorSchemeM3.light.primary,
      foregroundColor: AppColorSchemeM3.light.onPrimary,
      elevation: 0,
      centerTitle: true,
    ),

    // Textos
    textTheme: TextTheme(
      headlineLarge: TextStyle(fontSize: 32, fontWeight: FontWeight.bold, color: AppColorSchemeM3.light.primary),
      headlineMedium: TextStyle(fontSize: 28, fontWeight: FontWeight.w600, color: AppColorSchemeM3.light.primary),
      bodyLarge: TextStyle(fontSize: 16, color: AppColorSchemeM3.light.onBackground),
      bodyMedium: TextStyle(fontSize: 14, color: AppColorSchemeM3.light.onBackground.withOpacity(0.8)),
      labelLarge: TextStyle(fontSize: 14, fontWeight: FontWeight.w600, color: AppColorSchemeM3.light.secondary),
    ),

    // Iconos
    iconTheme: IconThemeData(color: AppColorSchemeM3.light.primary, size: 24),

    // Botones
    elevatedButtonTheme: ElevatedButtonThemeData(
      style: ElevatedButton.styleFrom(
        backgroundColor: AppColorSchemeM3.light.primary,
        foregroundColor: AppColorSchemeM3.light.onPrimary,
        padding: const EdgeInsets.symmetric(horizontal: 24, vertical: 14),
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      ),
    ),

    textButtonTheme: TextButtonThemeData(
      style: TextButton.styleFrom(
        foregroundColor: AppColorSchemeM3.light.secondary,
        padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 12),
      ),
    ),

    outlinedButtonTheme: OutlinedButtonThemeData(
      style: OutlinedButton.styleFrom(
        side: BorderSide(color: AppColorSchemeM3.light.secondary),
        foregroundColor: AppColorSchemeM3.light.secondary,

      color: AppColorSchemeM3.light.surface,

      shadowColor: AppColorSchemeM3.light.shadow,
    ),

    // TextFields
    inputDecorationTheme: InputDecorationTheme(
      filled: true,
      fillColor: AppColorSchemeM3.light.surfaceVariant,

        borderRadius: BorderRadius.circular(12),
        borderSide: BorderSide(color: AppColorSchemeM3.light.outline),
      ),
      focusedBorder: OutlineInputBorder(
        borderRadius: BorderRadius.circular(12),
        borderSide: BorderSide(color: AppColorSchemeM3.light.primary),

    ),

  static final ThemeData darkTheme = ThemeData(
